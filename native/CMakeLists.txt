cmake_minimum_required(VERSION 3.22)
project(learnime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
include("${CMAKE_CURRENT_SOURCE_DIR}/libime.cmake")
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    execute_process(COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/setup-sentencepiece.sh"
                    RESULT_VARIABLE _learnime_sp_result)
    if(NOT _learnime_sp_result EQUAL 0)
        message(FATAL_ERROR "Could not prepare the pinned SentencePiece dependency")
    endif()
    set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
    set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_TENSORFLOW_SHARED OFF CACHE BOOL "" FORCE)
    add_subdirectory(../third_party/sentencepiece sentencepiece EXCLUDE_FROM_ALL)
    foreach(_sp_tool spm_encode spm_decode spm_normalize spm_train spm_export_vocab)
        if(TARGET ${_sp_tool})
            set_property(TARGET ${_sp_tool} PROPERTY EXCLUDE_FROM_ALL TRUE)
        endif()
    endforeach()
endif()
# Portable libime decoder, shared by Android and iOS.
add_library(learnime-pinyin STATIC libime_pinyin.cpp)
target_include_directories(learnime-pinyin PUBLIC include)
target_link_libraries(learnime-pinyin PUBLIC learnime-libime-pinyin)
target_compile_options(learnime-pinyin PRIVATE -O2 -Wno-deprecated-declarations -Wno-format)
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(pinyin-smoke EXCLUDE_FROM_ALL ../scripts/pinyin-smoke.cpp)
    target_link_libraries(pinyin-smoke PRIVATE learnime-pinyin)
    target_compile_options(pinyin-smoke PRIVATE -UNDEBUG)
    add_executable(phrase-audit EXCLUDE_FROM_ALL ../scripts/phrase-audit.cpp)
    target_link_libraries(phrase-audit PRIVATE learnime-pinyin)
    add_executable(pinyin-contract EXCLUDE_FROM_ALL ../scripts/pinyin-contract.cpp)
    target_link_libraries(pinyin-contract PRIVATE learnime-pinyin)
    target_compile_options(pinyin-contract PRIVATE -UNDEBUG)
endif()
if(ANDROID)
    set(ZINNIA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/zinnia")
    add_library(learnime-zinnia STATIC
        ${ZINNIA_DIR}/recognizer.cpp ${ZINNIA_DIR}/sexp.cpp
        ${ZINNIA_DIR}/feature.cpp ${ZINNIA_DIR}/character.cpp)
    target_include_directories(learnime-zinnia PUBLIC ${ZINNIA_DIR})
    target_compile_definitions(learnime-zinnia PRIVATE HAVE_SYS_TYPES_H HAVE_SYS_STAT_H
        HAVE_FCNTL_H HAVE_STRING_H HAVE_SYS_MMAN_H HAVE_UNISTD_H HAVE_MMAP WORDS_LITENDIAN VERSION="0.06")
    add_library(learnime SHARED pinyin_jni.cpp sentencepiece_jni.cpp model_asset_jni.cpp handwriting_jni.cpp)
    target_include_directories(learnime PRIVATE include ../third_party/sentencepiece/src)
    target_link_libraries(learnime PRIVATE learnime-pinyin learnime-zinnia sentencepiece-static log)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(ZINNIA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/zinnia")
    add_library(learnime-handwriting STATIC handwriting.cpp
        ${ZINNIA_DIR}/recognizer.cpp ${ZINNIA_DIR}/sexp.cpp
        ${ZINNIA_DIR}/feature.cpp ${ZINNIA_DIR}/character.cpp)
    target_include_directories(learnime-handwriting PUBLIC include PRIVATE ${ZINNIA_DIR})
    target_compile_definitions(learnime-handwriting PRIVATE HAVE_SYS_TYPES_H HAVE_SYS_STAT_H
        HAVE_FCNTL_H HAVE_STRING_H HAVE_SYS_MMAN_H HAVE_UNISTD_H HAVE_MMAP VERSION="0.06")
    add_library(learnime-sentencepiece STATIC sentencepiece.cpp)
    target_include_directories(learnime-sentencepiece PUBLIC include PRIVATE ../third_party/sentencepiece/src)
    target_link_libraries(learnime-sentencepiece PRIVATE sentencepiece-static)
endif()
